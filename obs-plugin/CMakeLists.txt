cmake_minimum_required(VERSION 3.25)
project(clappa VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network)
add_executable(clappa-identity-test EXCLUDE_FROM_ALL identity-test.cpp)
target_include_directories(clappa-identity-test PRIVATE vendor "${NATIVE_DEPS}/include")
if(WIN32)
 enable_language(ASM_MASM)
 file(GLOB BLST_ASM ../vendor/blst/build/win64/*-x86_64.asm)
else()
 enable_language(ASM)
 set(BLST_ASM ../vendor/blst/build/assembly.S)
endif()
add_library(clappa_blst STATIC ../vendor/blst/src/server.c ${BLST_ASM})
set_target_properties(clappa_blst PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(clappa_blst PRIVATE __BLST_PORTABLE__)
add_library(clappa MODULE src/plugin.cpp src/native-service.cpp src/twitch-chat.cpp vendor/qrcodegen.cpp ../shared/beacon/quicknet.c)
target_include_directories(clappa PRIVATE ../vendor/blst/bindings ../shared/beacon)
target_link_libraries(clappa PRIVATE clappa_blst)
set(NATIVE_DEPS "" CACHE PATH "OBS native dependency bundle")
target_include_directories(clappa-identity-test PRIVATE "${NATIVE_DEPS}/include")
target_link_libraries(clappa-identity-test PRIVATE Qt6::Core Qt6::Gui Qt6::Network "${NATIVE_DEPS}/lib/mbedcrypto.lib")
if(WIN32)
 target_compile_options(clappa-identity-test PRIVATE /utf-8)
 target_compile_definitions(clappa-identity-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(clappa-identity-test PRIVATE bcrypt)
endif()
target_include_directories(clappa PRIVATE vendor "${NATIVE_DEPS}/include")
target_link_libraries(clappa PRIVATE "${NATIVE_DEPS}/lib/mbedtls.lib" "${NATIVE_DEPS}/lib/mbedx509.lib" "${NATIVE_DEPS}/lib/mbedcrypto.lib" "${NATIVE_DEPS}/lib/zlibstatic.lib")
if(WIN32)
 target_link_libraries(clappa PRIVATE ws2_32 crypt32 bcrypt)
 target_compile_definitions(clappa PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(clappa PRIVATE /bigobj /utf-8)
endif()
if(WIN32)
  set(OBS_SOURCE "" CACHE PATH "OBS source headers")
  set(OBS_IMPORT_DIR "" CACHE PATH "Import libraries matching installed OBS")
  target_include_directories(clappa PRIVATE "${OBS_SOURCE}/libobs" "${OBS_SOURCE}/frontend/api" "${CMAKE_CURRENT_SOURCE_DIR}/compat")
  target_link_libraries(clappa PRIVATE "${OBS_IMPORT_DIR}/obs.lib" "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
else()
  find_package(libobs REQUIRED)
  find_package(obs-frontend-api REQUIRED)
  target_link_libraries(clappa PRIVATE OBS::libobs OBS::obs-frontend-api)
endif()
target_link_libraries(clappa PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
target_compile_definitions(clappa PRIVATE QT_NO_KEYWORDS)
set_target_properties(clappa PROPERTIES PREFIX "")
qt_add_resources(clappa "mascot" PREFIX "/clappa" BASE "../assets/mascot-v4" FILES "../assets/mascot-v4/obs-tail.png" "../assets/mascot-v4/obs-crown.png")
qt_add_resources(clappa "twitch-brand" PREFIX "/clappa" BASE "../assets/twitch" FILES "../assets/twitch/glitch_flat_white.png")
qt_add_resources(clappa-identity-test "twitch-brand-test" PREFIX "/clappa" BASE "../assets/twitch" FILES "../assets/twitch/glitch_flat_white.png")
add_executable(clappa-timing-test EXCLUDE_FROM_ALL tile-timing-test.cpp)
add_executable(clappa-freshness-test EXCLUDE_FROM_ALL freshness-test.cpp ../shared/beacon/quicknet.c)
target_include_directories(clappa-freshness-test PRIVATE vendor ../vendor/blst/bindings ../shared/beacon)
target_link_libraries(clappa-freshness-test PRIVATE clappa_blst Qt6::Core)
target_compile_definitions(clappa-freshness-test PRIVATE QT_NO_KEYWORDS NOMINMAX)
add_executable(clappa-chat-test EXCLUDE_FROM_ALL chat-test.cpp)
target_link_libraries(clappa-chat-test PRIVATE Qt6::Core)
if(WIN32)
 target_compile_options(clappa-chat-test PRIVATE /utf-8)
endif()
